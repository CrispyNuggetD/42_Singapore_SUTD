EVAL_TMP=$(mktemp -d /tmp/push_swap_eval.XXXXXX)
printf 'Saved inputs, moves and reports: %s\n' "$EVAL_TMP"
CHECKER=./checker_linux
if ! test -s "$CHECKER" || ! test -x "$CHECKER"; then
    printf 'Reference checker is missing, empty or not executable. Restore it first.\n'
    exit 1

# All output redirects below target logs, never the checker executable.
EVAL_RUN=0

# Each run saves one answer, checks that exact answer and counts its moves.
# No second solver invocation is needed just to count instructions.
run_case()
{
    EVAL_RUN=$((EVAL_RUN + 1))
    local prefix="$EVAL_TMP/run_$EVAL_RUN"
    local started=$SECONDS solver_rc checker_rc count
    printf '%s\n' "$@" > "$prefix.input"
    ./push_swap "$@" > "$prefix.moves" 2> "$prefix.stderr" 3> "$prefix.debug"
    solver_rc=$?
    count=$(wc -l < "$prefix.moves")
    printf 'Run %s: n=%s, moves=%s, seconds=%s, solver exit=%s\n' \
        "$EVAL_RUN" "$#" "$count" "$((SECONDS - started))" "$solver_rc"
    if test "$solver_rc" -ne 0; then
        cat "$prefix.stderr"
        return 1
    fi
    "$CHECKER" "$@" < "$prefix.moves" > "$prefix.verdict" 2> "$prefix.checker_stderr"
    checker_rc=$?
    cat "$prefix.verdict"
    if test "$checker_rc" -ne 0 || test "$(cat "$prefix.verdict")" != OK; then
        printf 'FAIL: reference checker did not accept the answer\n'
        cat "$prefix.checker_stderr"
        return 1
    fi
    if test -s "$prefix.stderr" || test -s "$prefix.debug"; then
        printf 'Extra diagnostics found; check DEBUG before submission.\n'
    fi
    case $# in
        1|2|3) test "$count" -le 3 ;;
        5) test "$count" -le 12 ;;
        100) test "$count" -lt 700 ;;
        500) test "$count" -lt 5500 ;;
        *) return 0 ;;
    esac
    if test "$?" -ne 0; then
        printf 'FAIL: outside the maximum-score move band\n'
        return 1
    fi
}

expect_error()
{
    ./push_swap "$@" > "$EVAL_TMP/error.out" 2> "$EVAL_TMP/error.err"
    printf 'Error\n' > "$EVAL_TMP/expected_error"
    if test ! -s "$EVAL_TMP/error.out" && cmp -s "$EVAL_TMP/expected_error" "$EVAL_TMP/error.err"; then
        printf 'PASS: Error on stderr only\n'
    else
        printf 'FAIL: expected only Error on stderr\n'
        cat "$EVAL_TMP/error.out" "$EVAL_TMP/error.err"
        return 1
    fi
}

expect_silent()
{
    ./push_swap "$@" > "$EVAL_TMP/silent.out" 2> "$EVAL_TMP/silent.err" 3> "$EVAL_TMP/silent.debug"
    local rc=$?
    if test "$rc" -eq 0 && test ! -s "$EVAL_TMP/silent.out" \
        && test ! -s "$EVAL_TMP/silent.err" && test ! -s "$EVAL_TMP/silent.debug"; then
        printf 'PASS: silent success\n'
    else
        printf 'FAIL: expected silent success; exit=%s\n' "$rc"
        return 1
    fi
}

# 1. MEMORY / CRASH CHECKS
# Inspect every report: no invalid accesses, uninitialised-value errors or leaks.
# Exit 99 means Valgrind detected a memory error; invalid arguments may otherwise
# return the program's ordinary error status. Keep memory checks separate from
# performance timings. Apply the same Valgrind command to other inputs as needed.
valgrind --leak-check=full --show-leak-kinds=all --errors-for-leak-kinds=all \
    --error-exitcode=99 --log-file="$EVAL_TMP/valgrind_valid.log" \
    ./push_swap 1 5 2 4 3 > "$EVAL_TMP/valgrind.moves"
printf 'Valgrind run exit: %s\n' "$?"
"$CHECKER" 1 5 2 4 3 < "$EVAL_TMP/valgrind.moves"
cat "$EVAL_TMP/valgrind_valid.log"
valgrind --leak-check=full --show-leak-kinds=all --errors-for-leak-kinds=all \
    --error-exitcode=99 --log-file="$EVAL_TMP/valgrind_invalid.log" ./push_swap 1x
printf 'Valgrind invalid-input run exit: %s\n' "$?"
cat "$EVAL_TMP/valgrind_invalid.log"
# Additional heap-path check (>500), beyond the mirror's benchmark sizes.

    --error-exitcode=99 --log-file="$EVAL_TMP/valgrind_heap.log" \
    ./push_swap "${ARG[@]}" > "$EVAL_TMP/heap.moves"
printf 'Valgrind heap-path run exit: %s\n' "$?"
"$CHECKER" "${ARG[@]}" < "$EVAL_TMP/heap.moves"
cat "$EVAL_TMP/valgrind_heap.log"
